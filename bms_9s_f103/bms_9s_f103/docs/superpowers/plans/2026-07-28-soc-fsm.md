# SOC 状态机重构 — 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将 `soc_ocv.c` 从"增强型安时积分+OCV动态校正"重构为四状态机驱动（IDLE→DISCHARGE/CHARGE→POLARIZATION→IDLE）+ Q_max EMA 自学习 + SOC 安全余量映射。

**Architecture:** soc_ocv 保持为 App 层纯算法模块（零 BSP/RTOS 依赖）。状态机在 `soc_ocv_update()` 内部驱动，`soc_task` 仅做编排（锁→读→update→写回）。API 向后兼容调用者。

**Tech Stack:** STM32F103C8Tx + arm-none-eabi-gcc (C11, int64_t supported)

## Global Constraints

- 安时积分使用 `int64_t` 中间量（防止 <3600mA 时截断为 0）
- Q_max 学习仅在 DISCHARGE→POLARIZATION→IDLE 完整周期触发
- `idle_baseline` 仅在 POLARIZATION→IDLE 转换时设置一次
- POLARIZATION 中优先检查充/放中断条件
- SOC 映射: `display = (real - 30) × 1000 / 970, clamp[0, 1000]`
- 编译 0 警告 0 错误

```c
 * @brief   SOC 状态机模块 — 充放电检测 + Q_max 自学习 + 安全余量映射
 * @note    四状态: IDLE → DISCHARGE/CHARGE → POLARIZATION → IDLE
 *          真实 SOC 通过安时积分跟踪，极化消除后 OCV 修正
 *          Q_max 在深度放电后 EMA 自学习 (α=0.1)

 */

#ifndef __APP_SOC_OCV_H
#define __APP_SOC_OCV_H

#include "stm32f1xx_hal.h"

#ifdef __cplusplus
extern "C" {
#endif

/* ============================================================
 * 配置宏
 * ============================================================ */

#define DEFAULT_NOMINAL_MAH          20000L  /**< 默认标称容量 20Ah            */
#define SOC_DISPLAY_OFFSET              30   /**< 显示 SOC 底部偏移 (‰)         */
#define SOC_DISPLAY_SCALE              970   /**< 显示 SOC 映射分母              */
#define Q_MAX_EMA_DIVISOR               10   /**< Q_max EMA 除数 (α=0.1)        */
#define Q_MAX_LEARN_THRESHOLD_RATIO      2   /**< Q_max 学习: q_passed > Q/2    */
#define POLARIZATION_TIME_MS       1800000UL /**< 极化消除时间 30min             */
#define CHARGE_PLATEAU_DELTA_MV          5U  /**< 充电平台 ΔV 阈值 (mV)         */
#define CHARGE_PLATEAU_WINDOW_MS    300000UL /**< 充电平台检测窗口 5min          */
#define CHARGE_DETECT_CURRENT_MA       200   /**< 充电器接入检测电流 (mA)        */
#define CHARGE_DETECT_DEBOUNCE_S        10U  /**< 充电器接入去抖时间 (秒)         */
#define DISCHARGE_DETECT_CURRENT_MA    500   /**< 放电检测 |I| 阈值 (mA)         */
#define DISCHARGE_END_CURRENT_MA       300   /**< 放电结束 |I| 阈值 (mA)         */
#define DISCHARGE_END_CELL_MIN_MV     3000U  /**< 放电结束最低电压 (mV)          */
#define CHARGE_RISE_DETECT_MV          100   /**< 充电检测电压上升阈值 (mV)       */
#define IDLE_OCV_UPDATE_I_MAX_MA       200   /**< IDLE 中 OCV 更新的 |I| 上限   */

/* ============================================================
 * OCV-SOC 查表
 * ============================================================ */

#define SOC_OCV_TABLE_SIZE  21U

typedef struct {
    uint16_t ocv_mv;       /**< 单芯 OCV (mV)     */
    uint16_t soc_permil;   /**< 对应 SOC (0-1000) */
} ocv_soc_point_t;

/* ============================================================
 * 状态机类型
 * ============================================================ */

typedef enum {
    SOC_STATE_IDLE          = 0U,
    SOC_STATE_DISCHARGE     = 1U,
    SOC_STATE_CHARGE        = 2U,
    SOC_STATE_POLARIZATION  = 3U,
} soc_state_t;

/** @brief SOC 估算上下文 (状态机全部状态) */
typedef struct {
    /* 输出 */
    uint16_t     soc_display;       /**< 显示 SOC [0,1000], 底部 3% 余量 */
    uint16_t     soc_real;          /**< 真实 SOC [0,1000]                */
    uint16_t     ocv_mv;            /**< OCV 参考值 (mV)                  */
    int32_t      remaining_mah;     /**< 剩余容量 (mAh)                   */

    /* Q_max */
    int32_t      q_max_mah;         /**< 学习到的满容量 (mAh)             */
    int32_t      q_passed_mah;      /**< 本周期累计通过电量 (mAh)         */

    /* 状态机 */
    soc_state_t  state;             /**< 当前状态                         */
    uint16_t     idle_baseline_mv;  /**< IDLE 基准电压 (POL→IDLE 时更新) */
    uint16_t     pol_entry_mv;      /**< 进入 POLARIZATION 时的电压        */
    uint16_t     soc_at_entry;      /**< 进入充/放时的 SOC (‰)           */
    uint32_t     polarization_ms;   /**< 极化计时器 (ms)                  */
    uint8_t      from_discharge;    /**< 1=放电周期 (可触发 Q_max 学习)   */

    /* 充电器接入去抖 */
    uint8_t      charge_debounce_s; /**< 充电器接入累计秒数                */

    /* Q_max 学习历史 */
    uint16_t     last_soc_end;      /**< 上次充/放结束 SOC (‰)           */
    int32_t      last_q_passed;     /**< 上次充/放累计电量 (mAh)          */

    /* 充电平台检测 (5min 滑动窗口) */
    uint16_t     charge_peak_mv;    /**< 窗口内最高电压 (mV)              */
    uint16_t     charge_valley_mv;  /**< 窗口内最低电压 (mV)              */
    uint32_t     charge_window_ms;  /**< 窗口累计时间 (ms)                */

    uint8_t      initialized;       /**< 初始化标记                       */
} soc_ocv_ctx_t;

/* ============================================================
 * API
 * ============================================================ */

void          soc_ocv_init(int32_t nominal_mah);
void          soc_ocv_update(uint16_t cell_min_mv, int32_t current_ma,
                              uint32_t dt_ms);
uint16_t      soc_ocv_get_soc(void);             /* 显示 SOC [0,1000]     */
uint16_t      soc_ocv_get_real_soc(void);        /* 真实 SOC [0,1000]     */
uint16_t      soc_ocv_get_ocv(void);
int32_t       soc_ocv_get_remaining_mah(void);
soc_state_t   soc_ocv_get_state(void);           /* 调试: 当前状态        */
int32_t       soc_ocv_get_q_max(void);           /* 调试: 当前 Q_max      */
uint16_t      soc_ocv_lookup(uint16_t ocv_mv);   /* OCV→SOC 查表          */
const soc_ocv_ctx_t *soc_ocv_get_ctx(void);     /* 调试: 完整 ctx        */

#ifdef __cplusplus
}
#endif

#endif /* __APP_SOC_OCV_H */
```

```c
uint16_t soc_ocv_lookup(uint16_t ocv_mv)
{
    if (ocv_mv >= ocv_soc_table[0].ocv_mv) {
        return ocv_soc_table[0].soc_permil;
    }
    if (ocv_mv <= ocv_soc_table[SOC_OCV_TABLE_SIZE - 1U].ocv_mv) {
        return ocv_soc_table[SOC_OCV_TABLE_SIZE - 1U].soc_permil;
    }

    for (uint8_t i = 0U; i < (SOC_OCV_TABLE_SIZE - 1U); i++) {
        uint16_t v_hi = ocv_soc_table[i].ocv_mv;
        uint16_t v_lo = ocv_soc_table[i + 1U].ocv_mv;

        if (ocv_mv <= v_hi && ocv_mv >= v_lo) {
            int32_t soc_hi = (int32_t)ocv_soc_table[i].soc_permil;
            int32_t soc_lo = (int32_t)ocv_soc_table[i + 1U].soc_permil;
            int32_t dv     = (int32_t)(v_hi - v_lo);
            int32_t dsoc   = soc_hi - soc_lo;

            if (dv > 0) {
                return (uint16_t)(soc_hi - dsoc * (int32_t)(v_hi - ocv_mv) / dv);
            }
            return (uint16_t)soc_hi;
        }
    }
    return 500U;
}
```

在 getter 函数后添加:
```c
void soc_ocv_update(uint16_t cell_min_mv, int32_t current_ma,
                     uint32_t dt_ms)
{
    int32_t abs_i = (current_ma >= 0) ? current_ma : -current_ma;

    /* === 首次调用: OCV 初始化 === */
    if (ctx.initialized == 0U) {
        ctx.ocv_mv           = cell_min_mv;
        ctx.soc_real         = soc_ocv_lookup(cell_min_mv);
        ctx.remaining_mah    = (int32_t)((int64_t)ctx.q_max_mah
                                          * ctx.soc_real / 1000L);
        ctx.idle_baseline_mv = cell_min_mv;
        ctx.state            = SOC_STATE_IDLE;
        ctx.initialized      = 1U;
        return;
    }

    /* === 状态机调度 === */
    switch (ctx.state) {

    /* ---- IDLE: 等待充/放/静置 OCV 更新 ---- */
    case SOC_STATE_IDLE:
        if (current_ma < -(int32_t)DISCHARGE_DETECT_CURRENT_MA) {
            /* → DISCHARGE */
            ctx.state         = SOC_STATE_DISCHARGE;
            ctx.soc_at_entry  = ctx.soc_real;
            ctx.q_passed_mah  = 0;
            ctx.from_discharge = 1U;
        } else if (cell_min_mv > (uint16_t)(ctx.idle_baseline_mv
                                             + CHARGE_RISE_DETECT_MV)) {
            /* → CHARGE */
            ctx.state         = SOC_STATE_CHARGE;
            ctx.soc_at_entry  = ctx.soc_real;
            ctx.q_passed_mah  = 0;
            ctx.charge_peak_mv   = cell_min_mv;
            ctx.charge_valley_mv = cell_min_mv;
            ctx.charge_window_ms = 0U;
            ctx.from_discharge   = 0U;
        } else if (abs_i < IDLE_OCV_UPDATE_I_MAX_MA) {
            /* 静置: OCV 更新 SOC */
            ctx.soc_real      = soc_ocv_lookup(cell_min_mv);
            ctx.remaining_mah = (int32_t)((int64_t)ctx.q_max_mah
                                           * ctx.soc_real / 1000L);
        }
        /* idle_baseline 不在此更新 — 仅在 POL→IDLE 时设置 */
        break;

    /* ---- DISCHARGE: 安时积分 + 结束检测 + 交叉检测 ---- */
    case SOC_STATE_DISCHARGE: {
        int64_t delta_mah = (int64_t)current_ma * (int64_t)dt_ms / 3600000L;
        ctx.q_passed_mah  += (int32_t)((delta_mah < 0) ? -delta_mah : delta_mah);
        ctx.remaining_mah += (int32_t)delta_mah;

        /* 钳位 */
        if (ctx.remaining_mah < 0) { ctx.remaining_mah = 0; }
        if (ctx.remaining_mah > ctx.q_max_mah) {
            ctx.remaining_mah = ctx.q_max_mah;
        }

        ctx.soc_real = (uint16_t)((int64_t)ctx.remaining_mah
                                   * 1000L / ctx.q_max_mah);

        /* 交叉检测: 充电器接入 (current > +200mA, 持续 10s) */
        if (current_ma > CHARGE_DETECT_CURRENT_MA) {
            ctx.charge_debounce_s++;
            if (ctx.charge_debounce_s >= CHARGE_DETECT_DEBOUNCE_S) {
                ctx.state = SOC_STATE_CHARGE;
                ctx.soc_at_entry = ctx.soc_real;
                ctx.q_passed_mah = 0;
                ctx.charge_peak_mv   = cell_min_mv;
                ctx.charge_valley_mv = cell_min_mv;
                ctx.charge_window_ms = 0U;
                ctx.from_discharge   = 0U;
                ctx.charge_debounce_s = 0U;
                break;
            }
        } else {
            ctx.charge_debounce_s = 0U;
        }

        /* 放电结束条件 */
        if (abs_i < DISCHARGE_END_CURRENT_MA
            || cell_min_mv < DISCHARGE_END_CELL_MIN_MV) {
            ctx.state           = SOC_STATE_POLARIZATION;
            ctx.last_soc_end    = ctx.soc_real;
            ctx.last_q_passed   = ctx.q_passed_mah;
            ctx.polarization_ms = 0U;
            ctx.pol_entry_mv    = cell_min_mv;
        }
        break;
    }

    /* ---- CHARGE: 安时积分 + 平台检测 + 交叉检测 ---- */
    case SOC_STATE_CHARGE: {
        int64_t delta_mah = (int64_t)current_ma * (int64_t)dt_ms / 3600000L;
        ctx.q_passed_mah  += (int32_t)((delta_mah < 0) ? -delta_mah : delta_mah);
        ctx.remaining_mah += (int32_t)delta_mah;

        if (ctx.remaining_mah < 0) { ctx.remaining_mah = 0; }
        if (ctx.remaining_mah > ctx.q_max_mah) {
            ctx.remaining_mah = ctx.q_max_mah;
        }

        ctx.soc_real = (uint16_t)((int64_t)ctx.remaining_mah
                                   * 1000L / ctx.q_max_mah);

        /* 充电平台 5min 滑动窗口 */
        ctx.charge_window_ms += dt_ms;
        if (cell_min_mv > ctx.charge_peak_mv) {
            ctx.charge_peak_mv = cell_min_mv;
        }
        if (cell_min_mv < ctx.charge_valley_mv) {
            ctx.charge_valley_mv = cell_min_mv;
        }

        /* 交叉检测: 负载接入 (current < -500mA) */
        if (current_ma < -(int32_t)DISCHARGE_DETECT_CURRENT_MA) {
            ctx.state          = SOC_STATE_DISCHARGE;
            ctx.soc_at_entry   = ctx.soc_real;
            ctx.q_passed_mah   = 0;
            ctx.from_discharge = 1U;
            break;
        }

        /* 充电结束条件: 5min 内 ΔV < 5mV */
        if (ctx.charge_window_ms >= CHARGE_PLATEAU_WINDOW_MS) {
            uint16_t delta = (uint16_t)(ctx.charge_peak_mv
                                        - ctx.charge_valley_mv);
            if (delta < CHARGE_PLATEAU_DELTA_MV) {
                ctx.state           = SOC_STATE_POLARIZATION;
                ctx.last_soc_end    = ctx.soc_real;
                ctx.last_q_passed   = ctx.q_passed_mah;
                ctx.polarization_ms = 0U;
                ctx.pol_entry_mv    = cell_min_mv;
            }
            /* 窗口重置: 滑窗前进 */
            ctx.charge_peak_mv   = cell_min_mv;
            ctx.charge_valley_mv = cell_min_mv;
            ctx.charge_window_ms = 0U;
        }
        break;
    }

    /* ---- POLARIZATION: 计时 + 中断检测 + OCV 修正 + Q_max 学习 ---- */
    case SOC_STATE_POLARIZATION:
        /* 优先: 充/放中断检测 */
        if (current_ma < -(int32_t)DISCHARGE_DETECT_CURRENT_MA) {
            ctx.state          = SOC_STATE_DISCHARGE;
            ctx.soc_at_entry   = ctx.soc_real;
            ctx.q_passed_mah   = 0;
            ctx.from_discharge = 1U;
            break;
        }
        if (cell_min_mv > (uint16_t)(ctx.pol_entry_mv
                                      + CHARGE_RISE_DETECT_MV)) {
            ctx.state         = SOC_STATE_CHARGE;
            ctx.soc_at_entry  = ctx.soc_real;
            ctx.q_passed_mah  = 0;
            ctx.charge_peak_mv   = cell_min_mv;
            ctx.charge_valley_mv = cell_min_mv;
            ctx.charge_window_ms = 0U;
            ctx.from_discharge   = 0U;
            break;
        }

        /* 极化计时 */
        ctx.polarization_ms += dt_ms;

        if (ctx.polarization_ms >= POLARIZATION_TIME_MS) {
            /* OCV 修正 SOC */
            ctx.soc_real      = soc_ocv_lookup(cell_min_mv);
            ctx.remaining_mah = (int32_t)((int64_t)ctx.q_max_mah
                                           * ctx.soc_real / 1000L);

            /* Q_max 学习 (仅放电周期, 放电量 > Q_max/2, 分母 > 0) */
            if (ctx.from_discharge
                && ctx.last_q_passed > ctx.q_max_mah
                                       / Q_MAX_LEARN_THRESHOLD_RATIO) {
                uint16_t delta_soc = (uint16_t)(ctx.soc_at_entry
                                                - ctx.last_soc_end);
                if (delta_soc > 0U) {
                    int32_t q_new = (int32_t)((int64_t)ctx.last_q_passed
                                              * 1000L / delta_soc);
                    ctx.q_max_mah += (q_new - ctx.q_max_mah)
                                     / Q_MAX_EMA_DIVISOR;
                }
            }

            /* → IDLE, 记录基准电压 */
            ctx.state            = SOC_STATE_IDLE;
            ctx.idle_baseline_mv = cell_min_mv;
        }
        break;

    default:
        ctx.state = SOC_STATE_IDLE;
        break;
    }

    /* 更新 OCV 参考值 */
    ctx.ocv_mv = cell_min_mv;
}
```

```c
}
```
